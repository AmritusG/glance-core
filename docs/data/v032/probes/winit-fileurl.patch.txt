+/// The drag's file paths, read as Swift Glance reads them: each pasteboard item's
+/// `public.file-url` (no legacy `NSFilenamesPboardType` translation).
+fn dragged_paths(sender: &NSObject) -> Vec<std::path::PathBuf> {
+    let pb: Retained<NSPasteboard> = unsafe { msg_send_id![sender, draggingPasteboard] };
+    let items: Option<Retained<NSArray<AnyObject>>> = unsafe { msg_send_id![&pb, pasteboardItems] };
+    let mut out = Vec::new();
+    for item in items.iter().flat_map(|a| a.iter()) {
+        let s: Option<Retained<NSString>> = unsafe { msg_send_id![item, stringForType: NSPasteboardTypeFileURL] };
+        let Some(s) = s else { continue };
+        let url: Option<Retained<AnyObject>> = unsafe { msg_send_id![class!(NSURL), URLWithString: &*s] };
+        let Some(url) = url else { continue };
+        let path: Option<Retained<NSString>> = unsafe { msg_send_id![&url, path] };
+        if let Some(p) = path { out.push(std::path::PathBuf::from(p.to_string())); }
+    }
+    out
